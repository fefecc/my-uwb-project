## 二、实施步骤

### 步骤 1: 协议常量 + 类型定义（纯新增，不影响任何现有逻辑）

#### 1.1 uwb_protocol.h — 新增帧类型常量

在现有 `UwbFuncCode` 枚举之后增加：

```c
/* 现有: UWB_FUNC_DISCOVERY_REQ=0x20, DISCOVERY_RESP=0x21,
         UWB_FUNC_TWR_TAG_START=0x40, TWR_ANCHOR_RESP=0x41,
         UWB_FUNC_APP_DATA_CFG=0x70, APP_DATA_FRAG=0x71, APP_DATA_CTRL=0x72 */

/* plan-v4 新增 */
#define UWB_FUNC_APP_DATA_CTRL_RESP  0x73   // Anchor→Tag: ACK/WAIT/STOP 应答

/* DATA_CTRL 子类型 (ext_header[0]) */
#define UWB_DATA_CTRL_GET_INFO  0x01   // Tag 请求元信息
#define UWB_DATA_CTRL_PULL      0x02   // Tag 拉取分片
#define UWB_DATA_CTRL_DONE      0x03   // Tag 通知传输完成
#define UWB_DATA_CTRL_NACK      0x04   // Tag 请求重发某分片
#define UWB_DATA_CTRL_STOP      0x05   // Tag 终止传输

/* DATA_CTRL_RESP 应答类型 (ext_header[0]) */
#define UWB_DATA_RESP_ACK   0x00   // 确认
#define UWB_DATA_RESP_WAIT  0x01   // 等待（Anchor 忙）
#define UWB_DATA_RESP_STOP  0x02   // 终止

/* DATA_FRAG flags 位掩码 */
#define UWB_DATA_FRAG_FLAG_META  0x01   // bit0: meta 帧
```

**改动说明**: 仅新增 `#define` 和枚举值，不修改任何现有代码。

##### 2.3.3 帧构建函数（新增）

```c
/* 构建 DATA_CFG_REQ 帧（无 payload，无 ext_header） */
static int build_data_cfg_req(uint8_t *buf, uint16_t dst16, uint8_t seq);

/* 构建 DATA_CTRL 帧（ext_header: ctrl_type + frag_id + reserved 2B） */
static int build_data_ctrl(uint8_t *buf, uint16_t dst16, uint8_t seq,
                            uint8_t ctrl_type, uint8_t frag_id);

/* 构建 DATA_CTRL_RESP 帧（ext_header: resp_type + reserved 1B） */
static int build_data_resp_ack(uint8_t *buf, uint16_t dst16, uint8_t seq,
                                uint8_t resp_type);

/* 为 slot 中的 APP payload 加帧头（DATA_FRAG 类型） */
static void link_add_data_frag_header(int8_t slot_idx, uint16_t dst16,
                                       uint8_t frag_id, uint8_t total_frags,
                                       uint8_t flags, uint16_t payload_len);
```

注意：`build_data_cfg_req` 和 `build_data_ctrl` 这几种请求帧，因为 **没有 APP payload**（只有帧头），可以直接用 `tx_buf` 构建后通过 `PHY_CMD_TX_FRAME` 发送（复用 `fast_reply` 的 `tx_buf` 机制），不消耗 slot。也可统一用 slot：LINK 自己 Alloc(LINK_OWN)，构建帧，发送完后释放。**推荐第二种**，因为 LINK 已经负责 slot 回收，且 TAG 侧 slot 充足。

##### 2.3.4 drain_app_cmds() — 处理 APP 命令（新增函数）

```c
static void drain_app_cmds(void) {
    UwbLinkCmd cmd;
    while (UwbLinkCmd_Recv(&cmd, 0)) {  // 非阻塞取完
        switch (cmd.type) {
        case UWB_LINK_CMD_SEND_DATA_CFG:
            // 构建 CFG_REQ 帧
            // → 注册 slot 到挂起表 (TX_FRAME)
            // → PHY_CMD_TX_FRAME(has_pending_rx=true, rx_slot_count=1)
            // → g_data_xfer_state = UWB_LINK_DATA_TAG_WAIT_ACK
            break;

        case UWB_LINK_CMD_SEND_DATA_CTRL:
            // 构建 CTRL 帧 (ext_header: ctrl_type + frag_id)
            // → PHY_CMD_TX_FRAME(has_pending_rx=true, rx_slot_count=1)
            break;

        case UWB_LINK_CMD_SEND_FRAG:
            // APP 已写 payload 到 slot->data[]
            // → link_add_data_frag_header(slot_index, ...)
            // → 注册 slot 到挂起表 (TX_FRAME)
            // → PHY_CMD_TX_FRAME(slot_index, has_pending_rx=true, rx_slot_count=1)
            break;

        case UWB_LINK_CMD_SEND_ACK:
            // APP 已写 ACK payload 到 slot->data[]
            // → PHY_CMD_TX_FRAME(slot_index, has_pending_rx=false)
            // → g_data_xfer_state = UWB_LINK_DATA_IDLE
            break;

        case UWB_LINK_CMD_DATA_RESET:
            link_data_slot_free_all();
            g_data_xfer_state = UWB_LINK_DATA_IDLE;
            g_disc_count = 0;
            g_data_retry_slot = -1;
            // → 通知 APP: EVT(DATA_RESET_ACK)
            break;

        case UWB_LINK_CMD_DATA_ABORT:
            link_data_slot_free_all();
            g_data_xfer_state = UWB_LINK_DATA_IDLE;
            break;

        default:
            break;
        }
    }
}
```

##### 2.3.5 drain_phy_events() 扩展 — 解析数据帧

在现有 `drain_phy_events()` 函数的 `PHY_EVT_RX_SLOT_DONE` 处理中，在现有 `UWB_FUNC_DISCOVERY_RESP` 分支之后，增加数据帧的处理分支。

**关键原则**: 不修改现有的 DISC_RESP 处理逻辑，在同一个 `if-else` 链末尾追加新分支。

现有结构（伪代码）：
```c

    if (evt.slot_index >= 0) {
        uwb_slot_t *s = UwbSlots_Get(evt.slot_index);
        if (s->frame_type == UWB_FUNC_DISCOVERY_RESP) {
            // ... 现有 DISC_RESP 处理 ...
        }
        // plan-v4 在此追加 ↓
    }
    break;
```

追加内容：
```c
        /* plan-v4: 数据帧处理 */
        if (s->frame_type == UWB_FUNC_APP_DATA_CTRL_RESP) {
            // ACK/WAIT/STOP 应答帧
            // → 通知 APP: DATA_ACK 事件
            // → 如果是 DONE 的 ACK: g_data_xfer_state = IDLE, 释放重发备份 slot
            // → link_data_slot_unregister(备份slot)
        }
        else if (s->frame_type == UWB_FUNC_APP_DATA_FRAG) {
            // 收到分片帧
            // → 注册到挂起表 (RX_FRAME)
            // → 通知 APP: DATA_FRAG(slot_index, frag_id, total_frags, flags)
        }
        else if (s->frame_type == UWB_FUNC_APP_DATA_CFG) {
            // Anchor 收到 CFG_REQ
            // → 通知 APP: DATA_CFG(src_id) 事件
            // → g_data_xfer_state = UWB_LINK_DATA_ANCHOR_WAIT_CTRL
        }
        else if (s->frame_type == UWB_FUNC_APP_DATA_CTRL) {
            // Anchor 收到 CTRL 帧
            // → 解析 ext_header: ctrl_type + frag_id
            // → 通知 APP: DATA_CTRL(src_id, ctrl_type, frag_id) 事件
        }
```

同样在 `PHY_EVT_RX_FRAME`（IDLE 监听收帧）处理中追加相同分支。

在 `PHY_EVT_TX_DONE` 处理中增加：

case PHY_EVT_TX_DONE:

##### 2.3.6 主循环 — 2+1 交织调度

修改 `UwbLink_Task()` 主循环。现有结构：

```c
void UwbLink_Task(void *argument) {
    // ... init ...
    for (;;) {
        drain_phy_events();
        if (role == TAG && !disc_exchange_active) {
            link_tag_send_disc();
        }
        osDelay(1);
    }
}
```

修改为：

```c
void UwbLink_Task(void *argument) {
    // ... init ...
    for (;;) {
        drain_phy_events();
        drain_app_cmds();          // plan-v4 新增: 处理 APP 命令

        if (role == APP_ROLE_TAG) {
            if (g_data_xfer_state != UWB_LINK_DATA_IDLE) {
                // 数据传输进行中: 检查超时+重发
                link_tag_handle_data_xfer();
            } else if (!disc_exchange_active) {
                // 2+1 交织判断
                if (g_disc_count >= 2 && UwbLink_HasDataCmd()) {
                    g_disc_count = 0;
                    // drain_app_cmds 已处理命令并发起数据传输
                } else {
                    link_tag_send_disc();
                }
            }
        }
        // Anchor 侧: drain_phy_events + drain_app_cmds 已处理全部逻辑

        /* 定时扫描 */
        link_data_slot_scan_timeout();   // 1s 周期扫描过期 BACKUP

        osDelay(1);
    }
}
```

`RX_WINDOW_END` 事件处理中增加：
```c
case PHY_EVT_RX_WINDOW_END:
    // ... 现有逻辑 ...
    g_disc_count++;   // plan-v4: 2+1 交织计数
    break;
```

##### 2.3.7 link_tag_handle_data_xfer() — Tag 侧数据帧状态机（新增函数）

```c
static void link_tag_handle_data_xfer(void) {
    uint32_t now = HAL_GetTick();

    switch (g_data_xfer_state) {
    case UWB_LINK_DATA_TAG_WAIT_ACK:
        // 检查是否已收到 ACK（由 drain_phy_events 设置标志）
        // 超时 → 重发 (最多 3 次)
        //   retry_count++ → 重新 PHY_CMD_TX_FRAME(retry_slot)
        //   超过 3 次 → 通知 APP: DATA_FAIL → 状态回 IDLE
        break;

    default:
        break;
    }

    // 全局会话超时检查 (5s)
    if (now - g_data_xfer_start_ms > LINK_DATA_TAG_SESSION_TIMEOUT_MS) {
        link_data_slot_free_all();
        g_data_xfer_state = UWB_LINK_DATA_IDLE;
        // 通知 APP: DATA_FAIL
    }
}
```

##### 2.3.8 重发逻辑

```c
// 发送数据帧后保留 retry_slot，不释放
// 超时时重发同一 slot（不重新 Encode，零拷贝）

static bool link_data_retry(void) {
    if (g_data_retry_count >= LINK_DATA_RETRY_MAX) {
        // 超过重试上限
        return false;
    }
    if (g_data_retry_slot < 0) return false;

    // 15ms 间隔检查
    if (HAL_GetTick() - g_last_data_tx_ms < LINK_DATA_FRAME_INTERVAL_MS) {
        return true;  // 还没到间隔，等下一个周期
    }

    // 重发: 直接使用同一个 slot
    phy_cmd_t cmd = {
        .type = PHY_CMD_TX_FRAME,
        .slot_index = g_data_retry_slot,
        .tx_time = 0,
        .has_pending_rx = true,
        .rx_timeout_us = 2000,
        .rx_slot_count = 1,
    };
    UwbPhyCmd_Send(&cmd);
    g_data_retry_count++;
    g_last_data_tx_ms = HAL_GetTick();
    return true;
}
```

#### 3.2 新增 build_fast_reply_data_ack()

```c
/* uwb_phy.c 新增函数 */
static uint16_t build_fast_reply_data_ack(uint16_t src16, uint8_t seq) {
    UwbProtocolFrame frame;
    UwbProtocol_InitFrame(&frame, &g_phy.stack_cfg, src16, seq,
                          (uint8_t)UWB_FUNC_APP_DATA_CTRL_RESP);
    // DATA_CTRL_RESP: ext_header = resp_type(1B) + reserved(1B) = 2B
    frame.common.ext_header_len = 2;
    frame.ext_header[0] = UWB_DATA_RESP_ACK;
    frame.ext_header[1] = 0;  // reserved
    frame.common.payload_len = 0;

    uint16_t tx_len;
    if (!UwbProtocol_Encode(&frame, g_phy.tx_buf, sizeof(g_phy.tx_buf), &tx_len)) {
        return 0;
    }
    dwt_writetxdata(tx_len, g_phy.tx_buf, 0);
    dwt_writetxfctrl(tx_len, 0, 1);
    return tx_len;
}
```

#### 3.3 uwb_phy.c — 移除所有 UwbSlots_Free() 调用

**核心变更**: PHY 不再释放任何 slot，统一由 LINK 层通过事件携带的 `slot_index` 回收。

| 位置 | 当前行为 | 修改为 |
|------|---------|-------|
| TX_DONE FINISH（非 fast_reply） | `UwbSlots_Free(g_phy.tx_slot)` 然后 `slot_index = -1` | **删除 Free 调用**，`slot_index = g_phy.tx_slot` |
| irq_tx_error | `UwbSlots_Free(g_phy.tx_slot)` | **删除 Free 调用**，事件携带 slot_index |
| watchdog 超时 | `UwbSlots_Free(tx_slot/rx_slot)` | **删除 Free 调用**，事件携带 slot_index |

**LINK 层对应增加回收**：在 `drain_phy_events()` 的 `PHY_EVT_TX_DONE` 处理中增加 `UwbSlots_Free(evt.slot_index)`。

##### 4.2.3 drain_link_events() 扩展

在现有事件处理 `switch` 中追加 plan-v4 事件分支：**只增加 case，不修改现有 case**。

```c
static void drain_link_events(void) {
    UwbLinkAppEvent evt;
    while (xQueueReceive(UwbLink_AppEventQueue(), &evt, 0)) {
        /* RESETTING 状态下只接受 RESET_ACK */
        if (g_tag_sess.state == APP_SESS_RESETTING) {
            if (evt.type == UWB_LINK_APP_EVT_DATA_RESET_ACK) {
                session_reset_confirm();
            }
            if (evt.type == UWB_LINK_APP_EVT_DATA_FRAG) {
                UwbSlots_Free(evt.data.data_frag_t.slot_index);
            }
            continue;
        }

        switch (evt.type) {
        /* ====== 现有: 不改动 ====== */
        case UWB_LINK_APP_EVT_TWR_EXCHANGE:
            handle_twr_exchange(&evt.data.twr);
            // plan-v4 新增: 更新 MAC 表
            mac_table_update(evt.data.twr.anchor_id);
            break;
        case UWB_LINK_APP_EVT_NEIGHBOR_SEEN:
            app_log_info("NEIGHBOR: 0x%04X", evt.data.neighbor.short_id);
            mac_table_update(evt.data.neighbor.short_id);
            break;

        /* ====== plan-v4 新增 ====== */
        case UWB_LINK_APP_EVT_DATA_ACK:
            g_tag_sess.error_count = 0;
            handle_tag_ack(&evt);
            break;
        case UWB_LINK_APP_EVT_DATA_FRAG:
            g_tag_sess.error_count = 0;
            handle_tag_frag(&evt);
            break;
        case UWB_LINK_APP_EVT_DATA_ERROR:
            g_tag_sess.error_count++;
            if (g_tag_sess.error_count >= 2) {
                session_request_reset();
            }
            break;
        case UWB_LINK_APP_EVT_DATA_COMPLETE:
        case UWB_LINK_APP_EVT_DATA_FAIL:
            session_request_reset();
            break;
        default:
            break;
        }
    }
}
```

##### 4.2.5 MAC 表管理（新增）

```c
static void mac_table_update(uint16_t anchor_id) {
    // 查找已有条目或空闲位
    for (int i = 0; i < MAC_TABLE_SIZE; i++) {
        if (g_mac_table[i].valid && g_mac_table[i].anchor_id == anchor_id) {
            g_mac_table[i].last_seen_ms = HAL_GetTick();
            return;
        }
    }
    for (int i = 0; i < MAC_TABLE_SIZE; i++) {
        if (!g_mac_table[i].valid) {
            g_mac_table[i].valid = true;
            g_mac_table[i].anchor_id = anchor_id;
            g_mac_table[i].last_seen_ms = HAL_GetTick();
            g_mac_table[i].last_poll_ms = 0;
            return;
        }
    }
    // 表满: 替换最旧的条目
    // (实际实现时选择一个合适的淘汰策略)
}

static uint16_t mac_table_next(void) {
    static int round_robin = 0;
    uint32_t now = HAL_GetTick();
    for (int i = 0; i < MAC_TABLE_SIZE; i++) {
        int idx = (round_robin + i) % MAC_TABLE_SIZE;
        if (g_mac_table[idx].valid) {
            // 剔除超时条目 (3 分钟无通信)
            if (now - g_mac_table[idx].last_seen_ms > MAC_ENTRY_TIMEOUT_MS) {
                g_mac_table[idx].valid = false;
                continue;
            }
            round_robin = (idx + 1) % MAC_TABLE_SIZE;
            return g_mac_table[idx].anchor_id;
        }
    }
    return 0;
}

static void app_tag_poll_schedule(void) {
    if (g_tag_sess.state != APP_SESS_IDLE) return;
    if (HAL_GetTick() - g_last_session_end_ms < 1000) return;

    uint16_t anchor = mac_table_next();
    if (anchor == 0) return;

    // 发起新会话
    memset(&g_tag_sess, 0, sizeof(g_tag_sess));
    g_tag_sess.state = APP_SESS_WAIT_CFG_ACK;
    g_tag_sess.anchor_id = anchor;
    g_tag_sess.session_start_ms = HAL_GetTick();

    UwbLink_SendCmd(&(UwbLinkCmd){
        .type = UWB_LINK_CMD_SEND_DATA_CFG,
        .target_id = anchor,
        .slot_index = -1,
    });
}
```

##### 4.2.6 CRC16 工具函数（新增）

```c
static uint16_t crc16_update(uint16_t crc, uint8_t byte) {
    crc ^= (uint16_t)byte;
    for (int i = 0; i < 8; i++) {
        if (crc & 1)
            crc = (crc >> 1) ^ 0xA001;
        else
            crc >>= 1;
    }
    return crc;
}

static uint16_t crc16(const uint8_t *data, uint16_t len) {
    uint16_t crc = 0xFFFF;
    for (uint16_t i = 0; i < len; i++) {
        crc = crc16_update(crc, data[i]);
    }
    return crc;
}
```

### 3.1 零影响的改动

| 改动 | 原因 |
|------|------|
| `uwb_protocol.h` 新增常量和宏 | 纯新增，未被引用的 define 不产生任何代码 |
| `uwb_stack_types.h` 新增结构体 | 纯新增，未被引用的 typedef 不占内存 |
| `uwb_slots.h` 新增 `UWB_SLOT_APP_OWN` | 新枚举值，现有代码只判断 FREE/LINK_OWN/PHY_OWN |
| `uwb_buffers.h` 新增队列声明 | 纯新增，不影响现有两个队列 |
| `uwb_phy.h` 新增函数声明 | 纯新增 |

### 3.2 轻微影响的改动

| 改动 | 影响 | 缓解措施 |
|------|------|---------|
| `UWB_SLOT_NUM` 8→16 | 多分配 8×sizeof(uwb_slot_t) 内存 | slot 结构约 160 字节，8 个约 1.3KB，可接受 |
| `drain_phy_events()` 新增分支 | 数据帧走新分支，DISC 帧走旧分支 | 用 `if-else` 互斥，不会交叉 |
| `UwbLink_Task()` 主循环增加 `drain_app_cmds()` | 每次循环多一次非阻塞队列读取 | 队列空时只消耗一次 `xQueueReceive(0)`，开销极小 |
| `irq_rx_ok()` 新增 if 分支 | 增加一个 func_code 比较 | 与现有 DISC_REQ 分支并列，不影响热路径 |
| `UwbLinkAppEvent.data` 联合体扩展 | 联合体大小增加（取最大成员） | 新成员与现有 `UwbTwrExchange`（最大成员）大小相当或更小，不增加内存 |

### 3.3 需要验证的关键点

1. **PHY TX_DONE slot_index 传递**: PHY 不再在 FINISH 中 free slot，改为事件携带 slot_index。LINK 在收到 TX_DONE 事件后负责 free。**需要确认 LINK 在所有路径上都正确处理了 `PHY_EVT_TX_DONE` 的 slot_index**。

2. **重发 slot 不变性**: 重发时直接复用同一个 slot_index，确保 slot 在重试期间不被释放。**LINK 需要跟踪 retry_slot 并在所有退出路径释放它**。

3. **数据帧间隔 15ms**: 在有密集 Discovery 帧的情况下，数据帧发送前检查 `g_last_data_tx_ms`，不满足间隔则延迟到下一个周期。
